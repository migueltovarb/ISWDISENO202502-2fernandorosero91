' --- ENUMERACIONES ---

    EMPLEADO
    GERENTEPROYECTO
}

enum EstadoTarea {
    PENDIENTE
    ENPROCESO
    COMPLETADA
    RETRASADA
}

enum TipoNotificacion {
    NUEVATAREA
    ALERTAVENCIMIENTO
}

' --- CLASES DEL MODELO ---

    - nombre : String
    - correo : String
    - contraseña : String
    - rol : RolUsuario
    --
    + Usuario()
    + Usuario(nombre: String, correo: String, contraseña: String, rol: RolUsuario)
    + getNombre(): String
    + setNombre(nombre: String): void
    + getCorreo(): String
    + setCorreo(correo: String): void
    + getContraseña(): String
    + setContraseña(contraseña: String): void
    + getRol(): RolUsuario
    + setRol(rol: RolUsuario): void
    + cambiarContraseña(nuevaContraseña: String): boolean
    + toString(): String
}

class Proyecto {
    - nombre : String
    - descripcion : String
    - responsable : Usuario
    - fechaCreacion : Date
    - fechaFinalizacion : Date
    - tareas : List<Tarea>
    --
    + Proyecto()
    + Proyecto(nombre: String, descripcion: String, responsable: Usuario, fechaCreacion: Date, fechaFinalizacion: Date, tareas: List<Tarea>)
    + getNombre(): String
    + setNombre(nombre: String): void
    + getDescripcion(): String
    + setDescripcion(descripcion: String): void
    + getResponsable(): Usuario
    + setResponsable(responsable: Usuario): void
    + getFechaCreacion(): Date
    + setFechaCreacion(fechaCreacion: Date): void
    + getFechaFinalizacion(): Date
    + setFechaFinalizacion(fechaFinalizacion: Date): void
    + getTareas(): List<Tarea>
    + setTareas(tareas: List<Tarea>): void
    + editarProyecto(nombre: String, descripcion: String, responsable: Usuario, fechaFinalizacion: Date): void
    + eliminarProyecto(): void
    + toString(): String
}

class Tarea {
    - titulo : String
    - descripcion : String
    - responsable : Usuario
    - proyecto : Proyecto
    - fechaInicio : Date
    - fechaEntrega : Date
    - estado : EstadoTarea
    --
    + Tarea()
    + Tarea(titulo: String, descripcion: String, responsable: Usuario, proyecto: Proyecto, fechaInicio: Date, fechaEntrega: Date, estado: EstadoTarea)
    + getTitulo(): String
    + setTitulo(titulo: String): void
    + getDescripcion(): String
    + setDescripcion(descripcion: String): void
    + getResponsable(): Usuario
    + setResponsable(responsable: Usuario): void
    + getProyecto(): Proyecto
    + setProyecto(proyecto: Proyecto): void
    + getFechaInicio(): Date
    + setFechaInicio(fechaInicio: Date): void
    + getFechaEntrega(): Date
    + setFechaEntrega(fechaEntrega: Date): void
    + getEstado(): EstadoTarea
    + setEstado(estado: EstadoTarea): void
    + cambiarUsuario(usuario: Usuario): void
    + editarTarea(titulo: String, descripcion: String, fechaEntrega: Date, estado: EstadoTarea): void
    + eliminarTarea(): void
    + estaRetrasada(): boolean
    + diasRestantes(): int
    + toString(): String
}

    + getDestinatario(): Usuario
    + setDestinatario(destinatario: Usuario): void
    + getFechaEnvio(): Date
    + setFechaEnvio(fechaEnvio: Date): void
    + getTipo(): TipoNotificacion
    + setTipo(tipo: TipoNotificacion): void
    + enviarNotificacion(): void
    + toString(): String
}
